- 게임 종료 시 초기화 후 스폰으로 이동 (관전 모두 해제)

~~ DB ~~

games
game_id VARCHAR(36) PRIMARY KEY, game_type_id INT, channel_name VARCHAR(50), start_time DATETIME, end_time DATETIME, cancel_yn VARCHAR(1)

game_types
game_type_id INT PRIMARY KEY, type_name VARCHAR(20)

game_participants
game_id VARCHAR(36), player_uuid VARCHAR(36), PRIMARY KEY(game_id, player_uuid)

game_kills
game_id VARCHAR(36), killer_player_uuid VARCHAR(36), dead_player_uuid VARCHAR(36), kill_time DATETIME

game_winners
game_id VARCHAR(36), winner_player_uuid VARCHAR(36), PRIMARY KEY(game_id, winner_player_uuid)

game_player_stats
player_uuid VARCHAR(36), game_type_id INT, kill INT, death INT, assist INT, win INT, PRIMARY KEY(player_uuid, game_type_id)
